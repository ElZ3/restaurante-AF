CREATE DATABASE IF NOT EXISTS olivo_real;
USE olivo_real;

CREATE TABLE areas (
    id INT(10) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(60),
    descripcion VARCHAR(150),
    estado ENUM('activo', 'inactivo')
);

CREATE TABLE mesas (
    id INT(10) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    area_id INT(10) UNSIGNED,
    numero_mesa SMALLINT(5) UNSIGNED,
    capacidad TINYINT(3) UNSIGNED,
    estado ENUM('disponible', 'reservada', 'ocupada', 'fuera_servicio'),
    activo TINYINT(1),
    FOREIGN KEY (area_id) REFERENCES areas(id)
);

CREATE TABLE categorias (
    id INT(10) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(60),
    descripcion VARCHAR(150),
    estado ENUM('activo', 'inactivo')
);

CREATE TABLE productos (
    id INT(10) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    categoria_id INT(10) UNSIGNED,
    nombre VARCHAR(100),
    descripcion VARCHAR(255),
    precio DECIMAL(10,2),
    imagen VARCHAR(255),
    disponible TINYINT(1),
    creado_en TIMESTAMP,
    activo TINYINT(1),
    FOREIGN KEY (categoria_id) REFERENCES categorias(id)
);

CREATE TABLE usuarios (
    id INT(10) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100),
    correo VARCHAR(120),
    password_hash VARCHAR(255),
    rol ENUM('admin', 'mesero', 'cocina', 'cajero'),
    estado ENUM('activo', 'inactivo'),
    creado_en DATETIME,
    creado_por INT(10) UNSIGNED,
    FOREIGN KEY (creado_por) REFERENCES usuarios(id)
);

CREATE TABLE reservas (
    id INT(10) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    nombre_cliente VARCHAR(100),
    telefono_cliente VARCHAR(80),
    fecha_reserva DATE,
    hora TIME,
    personas TINYINT(3) UNSIGNED,
    mesa_id INT(10) UNSIGNED,
    estado ENUM('pendiente', 'confirmada', 'llego', 'atendida', 'cancelada'),
    observaciones VARCHAR(255),
    creado_por INT(10) UNSIGNED,
    creado_en DATETIME,
    FOREIGN KEY (mesa_id) REFERENCES mesas(id),
    FOREIGN KEY (creado_por) REFERENCES usuarios(id)
);

CREATE TABLE pedidos (
    id INT(10) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    mesa_id INT(10) UNSIGNED,
    usuario_id INT(10) UNSIGNED,
    reserva_id INT(10) UNSIGNED,
    fecha DATETIME,
    estado ENUM('borrador', 'pendiente', 'en_preparacion', 'lista', 'entregada', 'cancelada'),
    estado_cuenta ENUM('abierta', 'pagada', 'anulada'),
    total DECIMAL(10,2),
    fecha_pedido DATETIME,
    FOREIGN KEY (mesa_id) REFERENCES mesas(id),
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id),
    FOREIGN KEY (reserva_id) REFERENCES reservas(id)
);

CREATE TABLE detalle_pedido (
    id INT(10) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    pedido_id INT(10) UNSIGNED,
    producto_id INT(10) UNSIGNED,
    cantidad SMALLINT(5) UNSIGNED,
    precio_unitario DECIMAL(10,2),
    observaciones VARCHAR(255),
    FOREIGN KEY (pedido_id) REFERENCES pedidos(id),
    FOREIGN KEY (producto_id) REFERENCES productos(id)
);

CREATE TABLE pagos (
    id INT(10) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    pedido_id INT(10) UNSIGNED,
    usuario_id INT(10) UNSIGNED,
    monto DECIMAL(10,2),
    metodo ENUM('efectivo', 'tarjeta', 'transferencia', 'otro'),
    fecha DATETIME,
    referencia VARCHAR(80),
    token_unico CHAR(32),
    FOREIGN KEY (pedido_id) REFERENCES pedidos(id),
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);

CREATE TABLE historial_pedido (
    id INT(10) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    pedido_id INT(10) UNSIGNED,
    estado VARCHAR(30),
    usuario_id INT(10) UNSIGNED,
    fecha DATETIME,
    FOREIGN KEY (pedido_id) REFERENCES pedidos(id),
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);

CREATE TABLE ordenes (
    id INT(10) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    mesa_id INT(10) UNSIGNED,
    usuario_id INT(10) UNSIGNED,
    reserva_id INT(10) UNSIGNED,
    fecha DATETIME,
    estado ENUM('borrador', 'pendiente', 'en_preparacion', 'lista', 'entregada', 'cancelado'),
    estado_cuenta ENUM('abierta', 'pagada', 'anulada'),
    total DECIMAL(10,2)
);

CREATE TABLE configuracion (
    clave VARCHAR(50) PRIMARY KEY,
    valor VARCHAR(255)
);

CREATE TABLE intentos_login (
    id INT(10) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    correo VARCHAR(120),
    ip VARCHAR(45),
    fecha DATETIME
);
